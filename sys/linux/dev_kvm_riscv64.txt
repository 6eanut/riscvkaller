# Table 5 in https://docs.riscv.org/reference/isa/_attachments/riscv-privileged.pdf .
# Table 4 in https://github.com/riscv/riscv-aia/releases .
# CSRS supported: linux/arch/riscv/include/uapi/asm/kvm.h .
riscv64_csr = 0x100, 0x104, 0x105, 0x140, 0x141, 0x142, 0x143, 0x144, 0x180, 0x106, 0x10a, 0x10c, 0x150, 0x114, 0x154
